/// Where a publisher sends one message. Implement it over any client, or use the generated `nats/`
/// crate for a JetStream context. Called from the publisher's flusher thread and from the caller's
/// thread; it blocks until the broker has accepted the message or refused it.
pub trait Transport: Send + Sync + 'static {
    /// Why a publish failed.
    type Error: ::std::fmt::Display + Send + 'static;
    /// Publishes `payload` on `subject` and waits for the broker's answer.
    fn publish(&self, subject: &str, payload: &[u8]) -> ::std::result::Result<(), Self::Error>;
}

/// A publish that did not land: which subject, how many payloads, and why.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PublishError {
    /// The subject the message was for.
    pub subject: &'static str,
    /// How many payloads the message carried.
    pub items: usize,
    /// The encoder's or the transport's reason.
    pub message: String,
}

impl PublishError {
    fn new(subject: &'static str, items: usize, message: impl Into<String>) -> Self {
        Self {
            subject,
            items,
            message: message.into(),
        }
    }
}

impl ::std::fmt::Display for PublishError {
    fn fmt(&self, formatter: &mut ::std::fmt::Formatter<'_>) -> ::std::fmt::Result {
        write!(
            formatter,
            "publishing {} payload(s) on `{}` failed: {}",
            self.items, self.subject, self.message
        )
    }
}

impl ::std::error::Error for PublishError {}

/// Called with every failed background flush. The batch it names is dropped (`at_most_once`).
pub type ErrorCallback = ::std::sync::Arc<dyn Fn(PublishError) + Send + Sync>;

/// How a publisher reports what it cannot return.
#[derive(Clone, Default)]
pub struct PublisherOptions {
    /// Called with every failed background flush; `None` drops failures silently.
    pub on_error: Option<ErrorCallback>,
}

type Sink = ::std::sync::Arc<
    dyn Fn(&'static str, &[u8]) -> ::std::result::Result<(), String> + Send + Sync,
>;

fn sink<T: Transport>(transport: T) -> Sink {
    ::std::sync::Arc::new(move |subject, payload| {
        transport
            .publish(subject, payload)
            .map_err(|error| error.to_string())
    })
}

fn encode<P: ::serde::Serialize>(
    subject: &'static str,
    payload: &P,
) -> ::std::result::Result<Vec<u8>, PublishError> {
    ::serde_json::to_vec(payload).map_err(|error| PublishError::new(subject, 1, error.to_string()))
}

/// One JSON array of already-encoded payloads.
fn array(payloads: &[Vec<u8>]) -> Vec<u8> {
    let mut out = Vec::with_capacity(payloads.iter().map(Vec::len).sum::<usize>() + payloads.len() + 1);
    out.push(b'[');
    for (index, payload) in payloads.iter().enumerate() {
        if index > 0 {
            out.push(b',');
        }
        out.extend_from_slice(payload);
    }
    out.push(b']');
    out
}

#[derive(Default)]
struct Buffer {
    payloads: Vec<Vec<u8>>,
    oldest: Option<::std::time::Instant>,
    closed: bool,
}

/// One `array` channel: a buffer, and a thread that flushes it at `max_items` or `max_delay`.
#[allow(dead_code)]
struct Batcher {
    subject: &'static str,
    state: ::std::sync::Arc<(::std::sync::Mutex<Buffer>, ::std::sync::Condvar)>,
    sink: Sink,
    worker: ::std::sync::Mutex<Option<::std::thread::JoinHandle<()>>>,
}

#[allow(dead_code)]
impl Batcher {
    fn start(
        subject: &'static str,
        max_items: usize,
        max_delay: ::std::time::Duration,
        sink: Sink,
        on_error: Option<ErrorCallback>,
    ) -> Self {
        let state = ::std::sync::Arc::new((
            ::std::sync::Mutex::new(Buffer::default()),
            ::std::sync::Condvar::new(),
        ));
        let worker = {
            let state = state.clone();
            let sink = sink.clone();
            ::std::thread::spawn(move || loop {
                let (lock, ready) = &*state;
                let mut buffer = lock.lock().unwrap_or_else(::std::sync::PoisonError::into_inner);
                loop {
                    if buffer.closed || buffer.payloads.len() >= max_items {
                        break;
                    }
                    match buffer.oldest {
                        None => {
                            buffer = ready
                                .wait(buffer)
                                .unwrap_or_else(::std::sync::PoisonError::into_inner);
                        }
                        Some(oldest) => {
                            let waited = oldest.elapsed();
                            if waited >= max_delay {
                                break;
                            }
                            buffer = ready
                                .wait_timeout(buffer, max_delay - waited)
                                .unwrap_or_else(::std::sync::PoisonError::into_inner)
                                .0;
                        }
                    }
                }
                let payloads = ::std::mem::take(&mut buffer.payloads);
                buffer.oldest = None;
                let closed = buffer.closed;
                drop(buffer);
                if !payloads.is_empty() {
                    if let Err(message) = sink(subject, &array(&payloads)) {
                        if let Some(report) = &on_error {
                            report(PublishError::new(subject, payloads.len(), message));
                        }
                    }
                }
                if closed {
                    return;
                }
            })
        };
        Self {
            subject,
            state,
            sink,
            worker: ::std::sync::Mutex::new(Some(worker)),
        }
    }

    fn enqueue(&self, payload: Vec<u8>) -> ::std::result::Result<(), PublishError> {
        let (lock, ready) = &*self.state;
        let mut buffer = lock.lock().unwrap_or_else(::std::sync::PoisonError::into_inner);
        if buffer.closed {
            return Err(PublishError::new(self.subject, 1, "the publisher is closed"));
        }
        buffer.payloads.push(payload);
        if buffer.oldest.is_none() {
            buffer.oldest = Some(::std::time::Instant::now());
        }
        ready.notify_one();
        Ok(())
    }

    fn send(&self, payloads: Vec<Vec<u8>>) -> ::std::result::Result<(), PublishError> {
        if payloads.is_empty() {
            return Ok(());
        }
        (self.sink)(self.subject, &array(&payloads))
            .map_err(|message| PublishError::new(self.subject, payloads.len(), message))
    }

    fn flush(&self) -> ::std::result::Result<(), PublishError> {
        let payloads = {
            let (lock, _) = &*self.state;
            let mut buffer = lock.lock().unwrap_or_else(::std::sync::PoisonError::into_inner);
            buffer.oldest = None;
            ::std::mem::take(&mut buffer.payloads)
        };
        self.send(payloads)
    }

    fn stop(&self) {
        {
            let (lock, ready) = &*self.state;
            let mut buffer = lock.lock().unwrap_or_else(::std::sync::PoisonError::into_inner);
            buffer.closed = true;
            ready.notify_one();
        }
        let worker = self
            .worker
            .lock()
            .unwrap_or_else(::std::sync::PoisonError::into_inner)
            .take();
        if let Some(worker) = worker {
            let _ = worker.join();
        }
    }
}

impl Drop for Batcher {
    fn drop(&mut self) {
        self.stop();
    }
}
