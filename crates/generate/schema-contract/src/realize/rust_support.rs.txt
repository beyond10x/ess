/// Absence is independent of the field value, including a present JSON null.
#[derive(Debug, Clone, PartialEq)]
pub enum EssPresence<T> {
    Absent,
    Present(T),
}

impl<T> Default for EssPresence<T> {
    fn default() -> Self { Self::Absent }
}

impl<T> EssPresence<T> {
    pub fn is_absent(&self) -> bool { matches!(self, Self::Absent) }
}

impl<T: ::serde::Serialize> ::serde::Serialize for EssPresence<T> {
    fn serialize<S: ::serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
        match self {
            Self::Present(value) => ::serde::Serialize::serialize(value, serializer),
            Self::Absent => Err(::serde::ser::Error::custom("absent is not a JSON value")),
        }
    }
}

impl<'de, T: ::serde::Deserialize<'de>> ::serde::Deserialize<'de> for EssPresence<T> {
    fn deserialize<D: ::serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
        T::deserialize(deserializer).map(Self::Present)
    }
}

/// A required nullable field must still exist in its containing object.
pub fn ess_required<'de, D: ::serde::Deserializer<'de>, T: ::serde::Deserialize<'de>>(deserializer: D) -> Result<T, D::Error> {
    T::deserialize(deserializer)
}

/// The value set of JSON Schema false is empty.
#[derive(Debug, Clone, PartialEq, ::serde::Serialize, ::serde::Deserialize)]
pub enum EssNever {}
