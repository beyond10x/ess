/// A finite IEEE-754 binary64 value. Source decoding requires raw-capable JSON input.
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct EssBinary64(f64);

/// Construction refuses NaN and either infinity.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum EssBinary64Error { NonFinite }

impl ::std::fmt::Display for EssBinary64Error {
    fn fmt(&self, f: &mut ::std::fmt::Formatter<'_>) -> ::std::fmt::Result {
        f.write_str("Binary64 requires a finite value")
    }
}
impl ::std::error::Error for EssBinary64Error {}

impl EssBinary64 {
    pub fn new(value: f64) -> Result<Self, EssBinary64Error> {
        if value.is_finite() { Ok(Self(value)) } else { Err(EssBinary64Error::NonFinite) }
    }
    pub fn get(self) -> f64 { self.0 }
}

impl ::serde::Serialize for EssBinary64 {
    fn serialize<S: ::serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
        Self::new(self.0).map_err(::serde::ser::Error::custom)?;
        serializer.serialize_f64(self.0)
    }
}

impl<'de> ::serde::Deserialize<'de> for EssBinary64 {
    fn deserialize<D: ::serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
        let raw = <::std::boxed::Box<::serde_json::value::RawValue> as ::serde::Deserialize>::deserialize(deserializer)?;
        let token = raw.get().trim();
        if !matches!(token.as_bytes().first(), Some(b'-' | b'0'..=b'9')) {
            return Err(::serde::de::Error::custom("Binary64 requires a JSON number token"));
        }
        let value = token.parse::<f64>().map_err(::serde::de::Error::custom)?;
        Self::new(value).map_err(::serde::de::Error::custom)
    }
}
